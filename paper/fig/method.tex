% ---------- row 1 ----------
\node[io, text width=3.3cm, minimum height=1.55cm] (doc) at (1.75,0) {{\small\bfseries Long document $x$}\\[2pt]
  {\footnotesize \ldots{} the uuid for \emph{dusty-harbor} is \colorbox{cSalmon}{\texttt{\scriptsize 186c24bd-\ldots}} \ldots{} founded in 1995 \ldots}};
\node[mod=cPeach] (det) at (6.15,0) {{\small\bfseries\textsc{Detect}}\\[2pt]{\footnotesize flag high-entropy strings $\mathcal{H}(x)$ from the text; skip years and words}};
\node[mod=cAqua] (alloc) at (10.25,0) {{\small\bfseries\textsc{Allocate}}\\[2pt]{\footnotesize enlarge $\mathcal{H}(x)$ to $s_{\mathrm{hi}}$, shrink prose to $s_{\mathrm{lo}}$, keep $v(I)\le B$}};
\node[mod=cSalmon] (anc) at (14.35,0) {{\small\bfseries\textsc{Anchor}}\\[2pt]{\footnotesize append verbatim copies of $\mathcal{H}(x)$, paid by the saved budget}};
% ---------- row 2 ----------
\node[mod=cMist] (route) at (14.35,-2.95) {{\small\bfseries\textsc{Route}}\\[2pt]{\footnotesize $r=\frac{\text{anchor tokens}}{v(I)}$;\; image if $r\le\tau$, else text}};
\node[io, text width=2.7cm] (img) at (10.6,-2.35) {{\small\bfseries VLM input}\\{\footnotesize image + anchors}};
\node[io, text width=2.7cm] (txt) at (10.6,-3.55) {{\small\bfseries VLM input}\\{\footnotesize plain text (dense)}};

\caption{\textbf{Hazard-aware rendering.} \textsc{Detect} marks verbatim-risky strings in the text before rendering; \textsc{Allocate} enlarges only those strings and shrinks prose so that the image stays within the vision-token budget $B$ of uniform rendering (up to page rounding); \textsc{Anchor} spends the saved budget on verbatim copies of the flagged strings; \textsc{Route} sends documents whose anchors would be too expensive to plain text. The thumbnails show the same 32K-character document rendered uniformly and by our allocation. Every step is training-free and needs no access to the model.}
\label{fig:method}
